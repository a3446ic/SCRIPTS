DO BEGIN

--DECLARACIÓN DE VARIABLES
DECLARE V_NUM_POLIZA INT = 1000021;

--P1. CARTERA FILTRADO POR V_NUM_POLIZA
SELECT RAMO,IDMODALIDAD,NUM_POLIZA,NUM_ANUALIDAD,COD_MEDIADOR,COD_SUBCLAVE,ACTIVO,P_INTERMEDIACION,FECHA_INICIO,FECHA_FIN,MODIF_USER
FROM EXT.CARTERA 
WHERE NUM_POLIZA = V_NUM_POLIZA
ORDER BY NUM_ANUALIDAD
;

-- UPDATE EXT.RELASUJE SET POSITIONNAME = '4149-0000' WHERE NUM_POLIZA = 20502 AND COD_AGENT = 46170206;
--P2. RELASUJE FILTRADO POR V_NUM_POLIZA
SELECT * FROM EXT.RELASUJE 
WHERE NUM_POLIZA = V_NUM_POLIZA
;
--MEDIADORES
-- SELECT * FROM EXT.MODIFICAR_MEDIADOR WHERE IDHOST LIKE '%46170206%';

--QUERY FINAL
SELECT C.RAMO,C.ACTIVO,C.NUM_POLIZA,C.NUM_ANUALIDAD,C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE MEDIADOR
, LPAD(C.IDMODALIDAD,3,0) MODALIDAD_CARTERA
, CASE 
	WHEN R.NUM_POLIZA IS NULL THEN 'NO'
	ELSE 'SI'
END EN_RELASUJE
,R.POSITIONNAME,LPAD(R.MOD,3,0) MODALIDAD_RELASUJE
, CASE
	WHEN  R.NUM_POLIZA IS NULL THEN '-'
	ELSE CASE
			WHEN C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE = R.POSITIONNAME THEN 'SI'
			ELSE 'NO'
		END
END MISMO_MEDIADOR
, CASE	
	WHEN  R.NUM_POLIZA IS NULL THEN 'PÓLIZA NO EXISTE EN RELASUJE'
	ELSE CASE
			WHEN R.POSITIONNAME IS NULL THEN 'MEDIADOR NULL EN RELASUJE'
			WHEN C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE = R.POSITIONNAME THEN 'COINCIDE MEDIADOR'
			WHEN C.COD_MEDIADOR = SUBSTRING(R.POSITIONNAME,1,4) AND C.COD_SUBCLAVE != SUBSTRING(R.POSITIONNAME,5,9) THEN 'DISTINTA SUBCLAVE'
			WHEN (SELECT COUNT(*) FROM EXT.RELASUJE RJ WHERE RJ.NUM_POLIZA = C.NUM_POLIZA AND LPAD(C.IDMODALIDAD,3,0) = LPAD(RJ.MOD,3,0)) > 1 THEN 'VARIOS MEDIADORES EN RELASUJE'
			ELSE 'DISTINTO MEDIADOR'
		END
END TIPO_CASO
-- , C.MODIF_USER
-- , CASE
-- 		WHEN (C.MODIF_USER != 'CDL') THEN 'MANUAL'
-- 		ELSE '-'
-- 	END MODIFICADO
, CASE
		-- WHEN (C.MODIF_USER != 'CDL') THEN 'MANUAL'
		WHEN EXISTS(SELECT 1 FROM EXT.CARTERA CR WHERE CR.ACTIVO > 0 AND CR.NUM_POLIZA = C.NUM_POLIZA AND CR.MODIF_USER != 'CDL') THEN 'MANUAL'
		ELSE '-'
	END MODIFICADO
FROM EXT.CARTERA C 
LEFT JOIN EXT.RELASUJE R ON C.NUM_POLIZA = R.NUM_POLIZA AND LPAD(C.IDMODALIDAD,3,0) = LPAD(R.MOD,3,0)
LEFT JOIN (SELECT *
						,ROW_NUMBER() OVER (PARTITION BY CR.NUM_POLIZA ORDER BY CR.NUM_ANUALIDAD DESC) AS RN
					FROM EXT.CARTERA CR 
					WHERE  ACTIVO > 0
					--AND CR.NUM_POLIZA = V_NUM_POLIZA
					) CRT ON C.NUM_POLIZA = CRT.NUM_POLIZA AND CRT.COD_MEDIADOR = C.COD_MEDIADOR AND CRT.COD_SUBCLAVE = C.COD_SUBCLAVE AND CRT.RN = 1
LEFT JOIN (SELECT * FROM EXT.CARTERA CR WHERE CR.RAMO = 'CAUCION' AND CR.NUM_FIANZA IS NOT NULL) CRTC ON C.NUM_POLIZA = CRTC.NUM_POLIZA 
	AND CRTC.COD_MEDIADOR = C.COD_MEDIADOR AND CRTC.COD_SUBCLAVE = C.COD_SUBCLAVE AND CRTC.NUM_FIANZA = C.NUM_FIANZA
WHERE C.ACTIVO > 0
AND CRTC.NUM_FIANZA IS NOT NULL
-- FILTRO RAMO
AND 1 = (CASE 
			WHEN C.RAMO = 'CREDITO' AND CRT.RN=1 THEN 1
			WHEN C.RAMO = 'CAUCION' THEN 1
			ELSE 0
		END)
-- FILTRO EXISTEN EN RELASUJE
-- AND 1 = (CASE 
-- 			WHEN R.NUM_POLIZA IS NOT NULL THEN 1
-- 			ELSE 0
-- 		END
-- )
--FILTRO DISTINTOS MEDIADORES
AND 1 = (CASE
		WHEN C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE <> COALESCE(R.POSITIONNAME,'') THEN 1
		ELSE 0
		END
)

GROUP BY C.RAMO,C.ACTIVO,C.NUM_POLIZA,C.NUM_ANUALIDAD,C.COD_MEDIADOR,C.COD_SUBCLAVE,R.POSITIONNAME,LPAD(C.IDMODALIDAD,3,0),LPAD(R.MOD,3,0),R.NUM_POLIZA
--,C.MODIF_USER
ORDER BY NUM_POLIZA;

--QUERY FINAL FILTRADO POR PÓLIZA
SELECT C.RAMO,C.ACTIVO,C.NUM_POLIZA,C.NUM_ANUALIDAD,C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE MEDIADOR
, LPAD(C.IDMODALIDAD,3,0) MODALIDAD_CARTERA
, CASE 
	WHEN R.NUM_POLIZA IS NULL THEN 'NO'
	ELSE 'SI'
END EN_RELASUJE
,R.POSITIONNAME,LPAD(R.MOD,3,0) MODALIDAD_RELASUJE
, CASE
	WHEN  R.NUM_POLIZA IS NULL THEN '-'
	ELSE CASE
			WHEN C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE = R.POSITIONNAME THEN 'SI'
			ELSE 'NO'
		END
END MISMO_MEDIADOR
, CASE	
	WHEN  R.NUM_POLIZA IS NULL THEN 'PÓLIZA NO EXISTE EN RELASUJE'
	ELSE CASE
			WHEN R.POSITIONNAME IS NULL THEN 'MEDIADOR NULL EN RELASUJE'
			WHEN C.COD_MEDIADOR||'-'||C.COD_SUBCLAVE = R.POSITIONNAME THEN 'COINCIDE MEDIADOR'
			WHEN C.COD_MEDIADOR = SUBSTRING(R.POSITIONNAME,1,4) AND C.COD_SUBCLAVE != SUBSTRING(R.POSITIONNAME,5,9) THEN 'DISTINTA SUBCLAVE'
			WHEN (SELECT COUNT(*) FROM EXT.RELASUJE RJ WHERE RJ.NUM_POLIZA = C.NUM_POLIZA AND LPAD(C.IDMODALIDAD,3,0) = LPAD(RJ.MOD,3,0)) > 1 THEN 'VARIOS MEDIADORES EN RELASUJE'
			ELSE 'DISTINTO MEDIADOR'
		END
END TIPO_CASO
--, C.MODIF_USER
, CASE
		-- WHEN (C.MODIF_USER != 'CDL') THEN 'MANUAL'
		WHEN EXISTS(SELECT 1 FROM EXT.CARTERA WHERE ACTIVO > 0 AND NUM_POLIZA = C.NUM_POLIZA AND MODIF_USER != 'CDL') THEN 'MANUAL'
		ELSE '-'
	END MODIFICADO
FROM EXT.CARTERA C 
LEFT JOIN EXT.RELASUJE R ON C.NUM_POLIZA = R.NUM_POLIZA AND LPAD(C.IDMODALIDAD,3,0) = LPAD(R.MOD,3,0)
LEFT JOIN (SELECT * FROM EXT.CARTERA CR WHERE CR.RAMO = 'CAUCION' AND CR.NUM_POLIZA = V_NUM_POLIZA AND CR.NUM_FIANZA IS NOT NULL) CRTC ON C.NUM_POLIZA = CRTC.NUM_POLIZA 
	AND CRTC.COD_MEDIADOR = C.COD_MEDIADOR AND CRTC.COD_SUBCLAVE = C.COD_SUBCLAVE AND CRTC.NUM_FIANZA = C.NUM_FIANZA
LEFT JOIN (SELECT *
						,ROW_NUMBER() OVER (PARTITION BY CR.NUM_POLIZA ORDER BY CR.NUM_ANUALIDAD DESC) AS RN
					FROM EXT.CARTERA CR 
					WHERE  ACTIVO > 0
					AND CR.NUM_POLIZA = V_NUM_POLIZA
					AND RAMO = 'CREDITO'
					) CRT ON C.NUM_POLIZA = CRT.NUM_POLIZA AND CRT.COD_MEDIADOR = C.COD_MEDIADOR AND CRT.COD_SUBCLAVE = C.COD_SUBCLAVE 
WHERE C.ACTIVO > 0
 AND CRTC.NUM_FIANZA IS NOT NULL
AND 1 = (CASE 
			WHEN C.RAMO = 'CREDITO' AND CRT.RN=1 THEN 1
			WHEN C.RAMO = 'CAUCION' THEN 1
			ELSE 0
		END)
AND C.NUM_POLIZA = V_NUM_POLIZA
-- AND 1 = (CASE 
-- 			WHEN R.NUM_POLIZA IS NOT NULL THEN 1
-- 			ELSE 0
-- 		END
-- )

GROUP BY C.RAMO,C.ACTIVO,C.NUM_POLIZA,C.NUM_ANUALIDAD,C.COD_MEDIADOR,C.COD_SUBCLAVE,R.POSITIONNAME,LPAD(C.IDMODALIDAD,3,0),LPAD(R.MOD,3,0),R.NUM_POLIZA
--,C.MODIF_USER
;


SELECT * FROM EXT.CARTERA WHERE NUM_POLIZA = V_NUM_POLIZA AND RAMO = 'CAUCION' AND IDMODALIDAD = 98;
